Practical 3 — Temporal Database (PostgreSQL)
============================================

INSTALLATION & SETUP
Open Command Prompt and run: psql -U postgres (enter the password YOUR_PASSWORD).

COMMANDS / QUERIES

Query 1 — Check Connection
\conninfo

Query 2 — Create Temporal Database
CREATE DATABASE temporal_db;

Query 3 — Connect Database
\c temporal_db

Query 4 — Create Employee Table
CREATE TABLE employees ( emp_id SERIAL PRIMARY KEY, emp_name VARCHAR(50), position VARCHAR(50), salary NUMERIC(10,2), valid_from DATE NOT NULL, valid_to DATE NOT NULL );

Query 13 — Expire Employee Record

Query 14 — Display All Employees
SELECT * FROM employees ORDER BY emp_name, valid_from;

Query 15 — Find Current Employees
SELECT * FROM employees WHERE '2025-09-25' BETWEEN valid_from AND valid_to ORDER BY emp_name;

Query 16 — View Employee History
SELECT * FROM employees WHERE emp_name = 'Rahul Sharma' ORDER BY valid_from;

Query 17 — Find Date Employees
SELECT * FROM employees WHERE '2025-08-15' BETWEEN valid_from AND valid_to ORDER BY emp_name;

Query 18 — Display Salary History
SELECT emp_name, position, salary, valid_from, valid_to FROM employees ORDER BY emp_name, valid_from;
